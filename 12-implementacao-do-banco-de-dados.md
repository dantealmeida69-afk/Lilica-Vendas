# 12 implementacao do banco de dados

## 12.1 SGBD escolhido

🔲 **Proposta: PostgreSQL.** Gratuito, robusto, com transações, restrições (`CHECK`, `UNIQUE`, `FK`) e tipo `NUMERIC` para valores em reais. O grupo deve confirmar o SGBD definido pelo professor e ajustar o script se for outro.

## 12.2 Script de criação (DDL)

```sql
-- Lilica Vendas — criação do banco (PostgreSQL)

CREATE TABLE usuario (
  id_usuario    BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nome          VARCHAR(120) NOT NULL,
  email         VARCHAR(160) NOT NULL UNIQUE,
  senha_hash    VARCHAR(255) NOT NULL,
  perfil        VARCHAR(10)  NOT NULL CHECK (perfil IN ('ADMIN','VENDEDOR')),
  ativo         BOOLEAN      NOT NULL DEFAULT TRUE,
  data_cadastro TIMESTAMPTZ  NOT NULL DEFAULT now()
);

CREATE TABLE cliente (
  id_cliente    BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nome          VARCHAR(120) NOT NULL,
  telefone      VARCHAR(20),
  email         VARCHAR(160),
  data_cadastro TIMESTAMPTZ  NOT NULL DEFAULT now()
);

CREATE TABLE categoria (
  id_categoria BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nome         VARCHAR(80) NOT NULL UNIQUE
);

CREATE TABLE produto (
  id_produto     BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  id_categoria   BIGINT        NOT NULL REFERENCES categoria(id_categoria),
  nome           VARCHAR(120)  NOT NULL,
  descricao      TEXT,
  preco_venda    NUMERIC(10,2) NOT NULL CHECK (preco_venda >= 0),
  estoque_atual  INTEGER       NOT NULL DEFAULT 0 CHECK (estoque_atual >= 0),
  estoque_minimo INTEGER       NOT NULL DEFAULT 0 CHECK (estoque_minimo >= 0),
  ativo          BOOLEAN       NOT NULL DEFAULT TRUE
);

CREATE TABLE pedido (
  id_pedido   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  id_cliente  BIGINT REFERENCES cliente(id_cliente),
  id_usuario  BIGINT        NOT NULL REFERENCES usuario(id_usuario),
  data_pedido TIMESTAMPTZ   NOT NULL DEFAULT now(),
  status      VARCHAR(10)   NOT NULL DEFAULT 'ABERTO'
              CHECK (status IN ('ABERTO','PAGO','ENTREGUE','CANCELADO')),
  valor_total NUMERIC(12,2) NOT NULL DEFAULT 0 CHECK (valor_total >= 0),
  observacao  VARCHAR(300)
);

CREATE TABLE item_pedido (
  id_item        BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  id_pedido      BIGINT        NOT NULL REFERENCES pedido(id_pedido) ON DELETE CASCADE,
  id_produto     BIGINT        NOT NULL REFERENCES produto(id_produto),
  quantidade     INTEGER       NOT NULL CHECK (quantidade > 0),
  preco_unitario NUMERIC(10,2) NOT NULL CHECK (preco_unitario >= 0),
  subtotal       NUMERIC(12,2) NOT NULL,
  UNIQUE (id_pedido, id_produto),
  CHECK (subtotal = quantidade * preco_unitario)
);

CREATE TABLE pagamento (
  id_pagamento   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  id_pedido      BIGINT        NOT NULL REFERENCES pedido(id_pedido) ON DELETE CASCADE,
  forma          VARCHAR(10)   NOT NULL
                 CHECK (forma IN ('PIX','DINHEIRO','CREDITO','DEBITO','OUTRO')),
  valor          NUMERIC(12,2) NOT NULL CHECK (valor > 0),
  status         VARCHAR(10)   NOT NULL DEFAULT 'PENDENTE'
                 CHECK (status IN ('PENDENTE','CONFIRMADO','ESTORNADO')),
  data_pagamento TIMESTAMPTZ   NOT NULL DEFAULT now()
);

CREATE TABLE movimentacao_estoque (
  id_movimentacao   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  id_produto        BIGINT      NOT NULL REFERENCES produto(id_produto),
  id_pedido         BIGINT      REFERENCES pedido(id_pedido) ON DELETE SET NULL,
  tipo              VARCHAR(10) NOT NULL CHECK (tipo IN ('ENTRADA','SAIDA','ESTORNO','PERDA')),
  quantidade        INTEGER     NOT NULL CHECK (quantidade > 0),
  motivo            VARCHAR(200),
  data_movimentacao TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE registro_uso_ia (
  id_registro    BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  id_usuario     BIGINT REFERENCES usuario(id_usuario) ON DELETE SET NULL,
  funcionalidade VARCHAR(20) NOT NULL
                 CHECK (funcionalidade IN ('RESUMO_VENDAS','SUGESTAO_REPOSICAO','OUTRA')),
  modelo         VARCHAR(60) NOT NULL,
  tokens         INTEGER     NOT NULL CHECK (tokens >= 0),
  data_chamada   TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- Índices para as consultas mais frequentes
CREATE INDEX idx_pedido_data     ON pedido(data_pedido);
CREATE INDEX idx_pedido_cliente  ON pedido(id_cliente);
CREATE INDEX idx_item_produto    ON item_pedido(id_produto);
CREATE INDEX idx_mov_produto     ON movimentacao_estoque(id_produto, data_movimentacao DESC);
CREATE INDEX idx_pagamento_pedido ON pagamento(id_pedido);
```

## 12.3 Como o banco garante as regras de negócio

| Regra                           | Mecanismo                                                 |
| ------------------------------- | --------------------------------------------------------- |
| RN02 subtotal correto           | `CHECK (subtotal = quantidade * preco_unitario)`          |
| RN04 estoque nunca negativo     | `CHECK (estoque_atual >= 0)`                              |
| RN07 produto vendido não some   | Chave estrangeira sem cascata em `item_pedido.id_produto` |
| RN08 quantidade e preço válidos | `CHECK (quantidade > 0)` e `CHECK (preco_unitario >= 0)`  |
| RN09 produto único por pedido   | `UNIQUE (id_pedido, id_produto)`                          |
| RN12 pedido com vendedor        | `id_usuario NOT NULL` em `pedido`                         |
| RN14 e-mail único               | `UNIQUE` em `usuario.email`                               |
